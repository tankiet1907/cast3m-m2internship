% Draft mode: purple texts and boxes (include drafts and reviews)
% Review mode: yellow highlights and boxes (include reviews only)
% Final mode: clean (exclude drafts and reviews)

% ======================================================
% PACKAGES
% ======================================================

\usepackage{xcolor}
\usepackage{soul}
\usepackage{environ}
\usepackage{etoolbox}
\usepackage{tcolorbox}
\tcbuselibrary{breakable}

% ======================================================
% INTERNAL SWITCHES
% ======================================================

\newif\ifdraftmode
\newif\ifreviewmode

\ifdefstring{\thesismode}{DRAFT}{
  \draftmodetrue
  \reviewmodetrue
}{
  \ifdefstring{\thesismode}{REVIEW}{
    \draftmodefalse
    \reviewmodetrue
  }{
    \draftmodefalse
    \reviewmodefalse
  }
}

% ======================================================
% REVIEW SETTINGS
% ======================================================

\sethlcolor{yellow}

% Inline review
\newcommand{\review}[1]{%
  \ifreviewmode
    \hl{\textbf{[REVIEW]} #1}
  \fi
}

% Breakable review block
\NewEnviron{reviewblock}{
  \ifreviewmode
    \begin{tcolorbox}[
      breakable,
      colback=yellow!20,
      colframe=yellow!60!black,
      boxrule=0.8pt,
      arc=2pt,
      left=6pt,
      right=6pt,
      top=6pt,
      bottom=6pt
    ]
    \textbf{[REVIEW]}\\[4pt]
    \BODY
    \end{tcolorbox}
  \fi
}

% ======================================================
% DRAFT SETTINGS
% ======================================================

% Inline draft
\newcommand{\draft}[1]{%
  \ifdraftmode
    {\color{purple!80!black}\textbf{[DRAFT]} #1}
  \fi
}

% Breakable draft block
\NewEnviron{draftblock}{
  \ifdraftmode
    \begin{tcolorbox}[
      breakable,
      colback=purple!3,
      colframe=purple!30!black,
      boxrule=0.3pt,
      arc=2pt,
      left=6pt,
      right=6pt,
      top=6pt,
      bottom=6pt
    ]
    \textbf{[DRAFT]}\\[4pt]
    \BODY
    \end{tcolorbox}
  \fi
}
