% --- Grouped Nomenclature Configuration ---
\usepackage[intoc]{nomencl}
\usepackage{etoolbox}
\makenomenclature

% 1. Set a fixed width for the symbol column to ensure vertical alignment
\setlength{\nomlabelwidth}{2.5cm}
\renewcommand{\nomlabel}[1]{#1\hfil}

% 2. Define the unit command: forces units to the right margin
\newcommand{\nomunit}[1]{%
  \renewcommand{\nomentryend}{%
    \unskip\nobreak\hfill\nobreak% Remove trailing spaces and prevent accidental line breaks
    \makebox[2cm][l]{% Fixed-width box for units, left-aligned within the box
      \ifstrempty{#1}{}{[#1]}%
    }%
  }%
}

% 3. Configure group headers and vertical spacing
\renewcommand{\nomgroup}[1]{%
  % Adds an empty item and vertical space to "close" the previous group properly
  \item[] \vspace{1.5em} 
  \item[\bfseries
    \ifstrequal{#1}{A}{Finite Element notation}{%
    \ifstrequal{#1}{B}{Thermal symbols}{%
    \ifstrequal{#1}{C}{Hydraulic symbols}{%
    \ifstrequal{#1}{D}{Mechanical symbols}{%
    \ifstrequal{#1}{E}{Probabilistic symbols}{}}}}}]
  \vspace{0.5em} % Small gap after the group title
}
